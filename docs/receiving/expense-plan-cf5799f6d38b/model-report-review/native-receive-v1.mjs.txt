import assert from 'node:assert/strict';

import {previewExpensePlan,renderExpensePlan,identifySnapshot} from './compiled/src/expense-plan.js';
const root=new URL('./',import.meta.url).pathname;
const corpus=JSON.parse(await readFile(join(root,'scope-and-fixtures-v1.json'),'utf8'));
const groups=[],reports=[];
function deepFreeze(value){if(value&&typeof value==='object'){Object.values(value).forEach(deepFreeze);Object.freeze(value);}return value;}
async function group(name,fn){const row={name};try{row.evidence=await fn();row.passed=true;}catch(e){row.passed=false;row.error={message:String(e),stack:e.stack};}groups.push(row);}
async function make(id,ledger,plan,month='2026-10',currency='usd'){
 const before=JSON.stringify({ledger,plan});deepFreeze(ledger);deepFreeze(plan);
 const preview=previewExpensePlan(ledger,plan,month,currency);
 assert.equal(JSON.stringify({ledger,plan}),before,'Inputs remain unchanged');
 const ledgerBytes=Buffer.from(JSON.stringify(ledger)+'\n'),planBytes=Buffer.from(JSON.stringify(plan)+'\n');
 const ledgerPath=join(root,'inputs',id+'-ledger.json'),planPath=join(root,'inputs',id+'-plan.json');
 await writeFile(ledgerPath,ledgerBytes,{flag:'wx'});await writeFile(planPath,planBytes,{flag:'wx'});
 const identity={ledger:identifySnapshot(ledgerPath,ledgerBytes),plan:identifySnapshot(planPath,planBytes),generatedAt:'2026-10-08T23:40:00.000Z'};
 const html=renderExpensePlan(preview,identity);
 const inspection=html.match(/<script type="application\/json" id="expense-plan-data">([\s\S]*?)<\/script>/);
 assert(inspection);assert.deepEqual(JSON.parse(inspection[1]),JSON.parse(JSON.stringify({...preview,provenance:identity})));
 const path=join(root,'reports',id+'.html');await writeFile(path,html,{flag:'wx'});
 const item={id,path,htmlBytes:Buffer.byteLength(html),sha256:createHash('sha256').update(html).digest('hex'),preview,identity};
 reports.push(item);return item;
}
